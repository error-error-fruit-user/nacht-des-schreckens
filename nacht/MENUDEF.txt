// Nacht's own option; everything else is in Horde Hammer's options

OptionValue "Nacht_Zombies" {
    0, "$NACHT_MENU_ZOMBIES_NACHT"
    1, "$NACHT_MENU_ZOMBIES_ROGUES"
}

AddOptionMenu "OptionsMenu" {
    Option "$NACHT_MENU_ZOMBIES", "nacht_rogues_gallery", "Nacht_Zombies"
}

// and the simple options menu, which UZDoom shows instead while m_simpleoptions is on
AddOptionMenu "OptionsMenuSimple" {
    Option "$NACHT_MENU_ZOMBIES", "nacht_rogues_gallery", "Nacht_Zombies"
}
